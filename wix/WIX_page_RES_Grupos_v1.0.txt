// SCaD Residencial — Puente de página Wix para Gestión de Grupos v1.0
// Página del Panel donde exista un componente HTML con ID: #htmlGrupos

import {
  cargarGrupos,
  crearGrupo,
  actualizarGrupo,
  cambiarEstadoGrupo,
  buscarUsuarios,
  cargarIntegrantesGrupo,
  agregarIntegranteGrupo,
  quitarIntegranteGrupo
} from 'backend/resGrupos.web';

const HTML_ID = '#htmlGrupos';

function enviar(type, payload = {}) { $w(HTML_ID).postMessage({ type, ...payload }); }
function errorCode(err) {
  const raw=String(err?.message||err||'ERROR_DESCONOCIDO');
  const conocidos=['NO_AUTENTICADO','NO_AUTORIZADO','NOMBRE_REQUERIDO','GRUPO_ID_REQUERIDO','LIMITE_5_GRUPOS_ACTIVOS','NOMBRE_GRUPO_DUPLICADO','GRUPO_NO_ENCONTRADO','DATOS_INCOMPLETOS','USUARIO_NO_DISPONIBLE'];
  return conocidos.find(x=>raw.includes(x))||'ERROR_DESCONOCIDO';
}

async function procesar(msg={}) {
  switch(msg.action){
    case 'INIT': case 'REFRESH': { const data=await cargarGrupos(); enviar('STATE',{data}); return; }
    case 'CREATE_GROUP': { await crearGrupo(msg.data||{}); const data=await cargarGrupos(); enviar('STATE',{data,toast:'Grupo creado.'}); return; }
    case 'UPDATE_GROUP': { await actualizarGrupo(msg.data||{}); const data=await cargarGrupos(); enviar('STATE',{data,toast:'Grupo actualizado.'}); return; }
    case 'SET_GROUP_ACTIVE': { await cambiarEstadoGrupo(msg.grupoId,msg.activo===true); const data=await cargarGrupos(); enviar('STATE',{data,toast:msg.activo?'Grupo activado.':'Grupo desactivado.'}); return; }
    case 'SEARCH_USERS': { const data=await buscarUsuarios(msg.texto||''); enviar('USER_RESULTS',{data}); return; }
    case 'LOAD_MEMBERS': { const data=await cargarIntegrantesGrupo(msg.grupoId); enviar('MEMBERS',{data}); return; }
    case 'ADD_MEMBER': { await agregarIntegranteGrupo(msg.grupoId,msg.memberId); const data=await cargarIntegrantesGrupo(msg.grupoId); const state=await cargarGrupos(); enviar('MEMBERS',{data,toast:'Usuario agregado al grupo.'}); enviar('STATE',{data:state}); return; }
    case 'REMOVE_MEMBER': { await quitarIntegranteGrupo(msg.grupoId,msg.memberId); const data=await cargarIntegrantesGrupo(msg.grupoId); const state=await cargarGrupos(); enviar('MEMBERS',{data,toast:'Usuario retirado del grupo.'}); enviar('STATE',{data:state}); return; }
    default: throw new Error('ACCION_NO_RECONOCIDA');
  }
}

$w.onReady(()=>{
  $w(HTML_ID).onMessage(async(event)=>{
    try { await procesar(event.data||{}); }
    catch(err){ console.error('RES_GRUPOS_PANEL',err); enviar('ERROR',{code:errorCode(err)}); }
  });
});
